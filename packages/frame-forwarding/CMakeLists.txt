cmake_minimum_required(VERSION 3.20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.json" DSH_FRAME_FORWARDING_METADATA)
string(JSON DSH_FRAME_FORWARDING_VERSION GET "${DSH_FRAME_FORWARDING_METADATA}" version)
string(REGEX REPLACE "-.*$" "" DSH_FRAME_FORWARDING_CMAKE_VERSION "${DSH_FRAME_FORWARDING_VERSION}")
project(dsh_frame_forwarding VERSION ${DSH_FRAME_FORWARDING_CMAKE_VERSION} LANGUAGES CXX)
add_library(dsh_frame_forwarding_sdk STATIC src/producer.cc src/transport.cc)
target_compile_features(dsh_frame_forwarding_sdk PUBLIC cxx_std_17)
target_include_directories(dsh_frame_forwarding_sdk PUBLIC include PRIVATE src)
set_target_properties(dsh_frame_forwarding_sdk PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(dsh_frame_forwarding_sdk PRIVATE $<$<PLATFORM_ID:Windows>:/EHsc>)
if(APPLE)
  target_link_libraries(dsh_frame_forwarding_sdk PUBLIC "-framework IOSurface" "-framework CoreFoundation" bsm)
elseif(WIN32)
  target_compile_definitions(dsh_frame_forwarding_sdk PUBLIC _WIN32_WINNT=0x0602)
  target_link_libraries(dsh_frame_forwarding_sdk PUBLIC d3d11 dxgi advapi32)
else()
  message(FATAL_ERROR "DSH frame forwarding requires macOS or Windows")
endif()
install(TARGETS dsh_frame_forwarding_sdk ARCHIVE DESTINATION lib)
install(FILES include/dsh_frame_forwarding.h DESTINATION include)

option(DSH_FRAME_FORWARDING_WEBGPU "Build the native WebGPU producer adapter" OFF)
if(DSH_FRAME_FORWARDING_WEBGPU)
  if(NOT DSH_DAWN_SOURCE_DIR)
    message(FATAL_ERROR "Use scripts/build-webgpu.mjs to prepare the pinned Dawn sources")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(DAWN_BUILD_MONOLITHIC_LIBRARY STATIC CACHE STRING "" FORCE)
  set(DAWN_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory("${DSH_DAWN_SOURCE_DIR}" dawn)
  set_target_properties(webgpu_dawn PROPERTIES OUTPUT_NAME dawn)
  add_library(dsh_webgpu STATIC src/graphite-interop.cc)
  target_compile_features(dsh_webgpu PUBLIC cxx_std_20)
  target_include_directories(dsh_webgpu PUBLIC include)
  target_link_libraries(dsh_webgpu PUBLIC dsh_frame_forwarding_sdk dawn::webgpu_dawn)
  set_target_properties(dsh_webgpu PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(dsh_webgpu PRIVATE $<$<PLATFORM_ID:Windows>:/EHsc>)
  install(TARGETS dsh_webgpu ARCHIVE DESTINATION lib)
endif()
